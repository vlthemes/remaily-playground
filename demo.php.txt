// No mail in Playground: the message is still saved, and visitors see it went through
$remaily_demo_no_mail = static function ( $id ) {
	$config             = Config::get( $id );
	$config['settings'] = array_merge( (array) ( $config['settings'] ?? [] ), [ 'skip_mail' => true, 'store_entries' => true ] );
	Config::save( $id, $config );
};

// title => [ intro, form ID ]
$remaily_demo_made = [];

if ( class_exists( Presets::class ) ) {
	$id = Presets::create( 'every-field' );

	if ( $id && !is_wp_error( $id ) ) {
		$remaily_demo_no_mail( $id );
		$remaily_demo_made[ get_the_title( $id ) ] = [ 'Every field type at once: a phone with its country and mask, a country list with flags, a date, a time, a slider, stars, "type your email again", an output that greets you, a character count, a group that shows for a company and rows guests add themselves.', $id ];
	}
} else {
	foreach ( $remaily_demo_forms as $title => [ $intro, $template, $config ] ) {
		$id = wp_insert_post( [
			'post_type'    => PostType::NAME,
			'post_status'  => 'publish',
			'post_title'   => $title,
			'post_content' => $template,
		] );

		if ( $id && !is_wp_error( $id ) ) {
			Config::save( $id, $config );
			$remaily_demo_no_mail( $id );
			$remaily_demo_made[ $title ] = [ $intro, $id ];
		}
	}
}

$remaily_demo_content = '<!-- wp:paragraph --><p>A Re:Maily form. Send it: Playground has no mail, so the message goes straight to Contact Form → Entries. Open the form in Contact Form → Contact Forms to see how it is made, or click Add New to start from another ready form: a quote in steps, a callback, feedback.</p><!-- /wp:paragraph -->' . "\n";

foreach ( $remaily_demo_made as $title => [ $intro, $id ] ) {
	// Several forms (Re:Maily before 1.0.1) get a heading each
	$remaily_demo_content .= sprintf(
		"%s<!-- wp:paragraph --><p>%s</p><!-- /wp:paragraph -->\n<!-- wp:shortcode -->[remaily id=\"%s\"]<!-- /wp:shortcode -->\n",
		count( $remaily_demo_made ) > 1 ? '<!-- wp:heading --><h2 class="wp-block-heading">' . esc_html( $title ) . '</h2><!-- /wp:heading -->' . "\n" : '',
		esc_html( $intro ),
		esc_attr( Form::hash_of( $id ) )
	);
}

// A page with every form, the site's home page
$remaily_demo_page = wp_insert_post( [
	'post_type'    => 'page',
	'post_status'  => 'publish',
	'post_title'   => 'Re:Maily demo',
	'post_name'    => 'remaily-demo',
	'post_content' => $remaily_demo_content,
] );

if ( $remaily_demo_page && !is_wp_error( $remaily_demo_page ) ) {
	update_option( 'show_on_front', 'page' );
	update_option( 'page_on_front', $remaily_demo_page );
}

// A demo opens where the blueprint says: no welcome redirect, no request for a review
delete_transient( 'remaily_welcome_redirect' );
update_option( 'remaily_review', 'done', false );

update_option( 'permalink_structure', '/%postname%/' );
